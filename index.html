    (function() {

      var API_URL = 'https://script.google.com/macros/s/AKfycbzL0QyUQXRurn9EArHWyvZXyUAH7yYBnrMy_eKV-Xf2GpDo8Y5bkEpsHUAdEFR1T_puUw/exec';


      var playlistTotale = [];

      var playlistIndex = 0;

      var stepTimer = null;

      var zoneTimers = [];

      var globalCorsiData = [];


      fetchData();


      function mostraLoader(mostra) {

        var loader = document.getElementById("ledwallLoader");

        if (loader) {

          if (mostra) loader.classList.remove("d-none");

          else loader.classList.add("d-none");

        }

      }


      function fetchData() {

        if (API_URL.indexOf("INSERISCI_QUI") !== -1) {

          console.error("URL API non configurato.");

          return;

        }

        mostraLoader(true);

        fetch(API_URL)

          .then(function(res) { return res.json(); })

          .then(function(data) {

            elaboraDatiLedwall(data, true);

            mostraLoader(false);

          })

          .catch(function(err) {

            mostraLoader(false);

            console.error("Errore fetch ledwall:", err);

          });

      }


      function aggiornaPlaylistSilenziosamente(callback) {

        if (API_URL.indexOf("INSERISCI_QUI") !== -1) {

          if (typeof callback === 'function') callback();

          return;

        }

        fetch(API_URL)

          .then(function(res) { return res.json(); })

          .then(function(data) {

            elaboraDatiLedwall(data, false);

            if (typeof callback === 'function') callback();

          })

          .catch(function(err) {

            console.error("Errore aggiornamento silenzioso:", err);

            if (typeof callback === 'function') callback();

          });

      }


      function elaboraDatiLedwall(data, avviaSubito) {

        if (!data) return;

        globalCorsiData = data.corsiTable || [];

        var haSerie = Array.isArray(data.serieAttive) && data.serieAttive.length > 0;

        var stage = document.getElementById("ledwallStage");


        if (!haSerie) {

          if (stage) {

            stage.innerHTML = '<div class="m-auto text-center text-dark p-5 w-100 h-100 d-flex flex-column justify-content-center align-items-center"><i class="ti ti-alert-triangle display-3 mb-3 text-danger"></i><h3 class="fw-bold mb-2">Nessuna serie attiva trovata</h3><p class="fs-4 mb-0 text-muted">Crea e attiva almeno una serie nel foglio Google.</p></div>';

          }

          return;

        }


        var nuovaPlaylist = [];

        data.serieAttive.forEach(function(ser) {

          if (ser && ser.sequence) {

            ser.sequence.forEach(function(step) {

              nuovaPlaylist.push(step);

            });

          }

        });


        if (nuovaPlaylist.length > 0) {

          playlistTotale = nuovaPlaylist;

          if (avviaSubito) {

            renderizzaStepCorrente();

          }

        }

      }


      function pulisciTimer() {

        if (stepTimer) clearTimeout(stepTimer);

        zoneTimers.forEach(function(t) { clearTimeout(t); });

        zoneTimers = [];

      }


      function getGridClassByLayout(layout) {

        var l = (layout || "").toLowerCase().trim();

        if (l.indexOf("1/2 height sx - up") !== -1 || l.indexOf("1/2 height sx-up") !== -1) return "grid-half-height-sx-up";

        if (l.indexOf("1/2 height sx - down") !== -1 || l.indexOf("1/2 height sx-down") !== -1) return "grid-half-height-sx-down";

        if (l.indexOf("1/2 height dx - up") !== -1 || l.indexOf("1/2 height dx-up") !== -1) return "grid-half-height-dx-up";

        if (l.indexOf("1/2 height dx - down") !== -1 || l.indexOf("1/2 height dx-down") !== -1) return "grid-half-height-dx-down";

        if (l.indexOf("full-height sx") !== -1 || l.indexOf("full-height left") !== -1 || l.indexOf("sx full height") !== -1) return "grid-full-height-sx";

        if (l.indexOf("full-height dx") !== -1 || l.indexOf("full-height right") !== -1 || l.indexOf("dx full height") !== -1) return "grid-full-height-dx";

        if (l.indexOf("1/2 height sx") !== -1 || l.indexOf("half height sx") !== -1) return "grid-half-height-sx";

        if (l.indexOf("1/2 height dx") !== -1 || l.indexOf("half height dx") !== -1) return "grid-half-height-dx";

        if (l.indexOf("full-lenght up") !== -1 || l.indexOf("full-length up") !== -1 || l === "up") return "grid-full-lenght-up";

        if (l.indexOf("full-lenght down") !== -1 || l.indexOf("full-length down") !== -1 || l === "down") return "grid-full-lenght-down";

        return "grid-fullscreen";

      }


      function renderizzaStepCorrente() {

        var stage = document.getElementById("ledwallStage");

        if (!stage || playlistTotale.length === 0) {

          setTimeout(renderizzaStepCorrente, 2000);

          return;

        }


        pulisciTimer();


        var currentStep = playlistTotale[playlistIndex % playlistTotale.length];

        var schermiDelStep = currentStep.schermi || [];


        var zoneMap = {};

        schermiDelStep.forEach(function(item) {

          var gridClass = getGridClassByLayout(item.layout);

          if (!zoneMap[gridClass]) zoneMap[gridClass] = [];

          zoneMap[gridClass].push(item);

        });


        var zoneKeys = Object.keys(zoneMap);

        if (zoneMap["grid-fullscreen"]) zoneKeys = ["grid-fullscreen"];


        stage.innerHTML = zoneKeys.map(function(gridClass) {

          return '<div id="zone-' + gridClass + '" class="grid-card-item ' + gridClass + '"></div>';

        }).join("");


        var maxDurataStepMs = (Number(currentStep.durata) || 10) * 1000;


        zoneKeys.forEach(function(gridClass) {

          var listaSchermiZona = zoneMap[gridClass];

          var tempoZonaMs = listaSchermiZona.reduce(function(acc, s) { return acc + ((Number(s.tempo) || 10) * 1000); }, 0);

          if (tempoZonaMs > maxDurataStepMs) maxDurataStepMs = tempoZonaMs;

          avviaRotazioneZona(gridClass, listaSchermiZona, 0);

        });


        pianificaProssimoStep(maxDurataStepMs);

      }


      function avviaRotazioneZona(gridClass, listaSchermi, index) {

        var el = document.getElementById("zone-" + gridClass);

        if (!el || !listaSchermi || !listaSchermi.length) return;


        var item = listaSchermi[index % listaSchermi.length];


        el.innerHTML = "";

        el.style.backgroundColor = item.bg || "#1a1b1e";


        if (item.media) {

          var mediaUrl = item.media.trim();

          var imgDiv = document.createElement("div");

          imgDiv.className = "preview-media-box";

          imgDiv.style.backgroundImage = "url(" + JSON.stringify(mediaUrl) + ")";

          el.appendChild(imgDiv);

        }


        var tipoSchermo = (item.tipo || "").toLowerCase();

        var isMeteo = tipoSchermo.indexOf("meteo") !== -1;

        var isCorsi = tipoSchermo.indexOf("corsi") !== -1;

        var textAlign = item.contentAlign || "center";


        var contentBox = document.createElement("div");

        contentBox.className = "preview-content-box";

        if ((isMeteo || isCorsi) && !item.titolo && !item.sottotitolo && !item.contenuto) {

          contentBox.style.justifyContent = "center";

          contentBox.style.alignItems = "center";

        } else {

          contentBox.style.textAlign = textAlign;

          contentBox.style.alignItems = textAlign === "left" ? "flex-start" : (textAlign === "right" ? "flex-end" : "center");

        }


        var innerWrapper = document.createElement("div");

        innerWrapper.className = "w-100";

        innerWrapper.style.maxWidth = (isMeteo || isCorsi) ? "1100px" : "95%";


        if (item.titolo) {

          var tDiv = document.createElement("div");

          tDiv.className = "fw-bold mb-2";

          tDiv.style.color = item.titoloColor || "#ffffff";

          tDiv.innerHTML = item.titolo;

          innerWrapper.appendChild(tDiv);

        }


        if (item.sottotitolo) {

          var sDiv = document.createElement("div");

          sDiv.className = "mb-3 fs-4";

          sDiv.style.color = item.sottotitoloColor || "#adb5bd";

          sDiv.innerHTML = item.sottotitolo;

          innerWrapper.appendChild(sDiv);

        }


        if (item.contenuto) {

          var cDiv = document.createElement("div");

          cDiv.className = "fs-4 mb-0";

          cDiv.style.color = item.contenutoColor || "#ffffff";

          cDiv.innerHTML = item.contenuto;

          innerWrapper.appendChild(cDiv);

        }


        if (isMeteo) {

          var meteoDiv = document.createElement("div");

          meteoDiv.id = "meteo-widget-" + gridClass;

          meteoDiv.className = "meteo-light-wrapper";

          meteoDiv.textContent = "Caricamento meteo Frascati...";

          innerWrapper.appendChild(meteoDiv);

        }


        if (isCorsi) {

          var corsiDiv = document.createElement("div");

          corsiDiv.id = "corsi-widget-" + gridClass;

          corsiDiv.className = "corsi-ledwall-wrapper";

          corsiDiv.innerHTML = '<div class="text-center text-muted fs-4 py-4">Caricamento corsi in corso...</div>';

          innerWrapper.appendChild(corsiDiv);

        }


        contentBox.appendChild(innerWrapper);

        el.appendChild(contentBox);


        if (isMeteo) caricaMeteoDettagliato("meteo-widget-" + gridClass);

        if (isCorsi) caricaCorsiArrampicata("corsi-widget-" + gridClass, globalCorsiData);


        if (listaSchermi.length > 1) {

          var durataItemMs = (Number(item.tempo) || 10) * 1000;

          var t = setTimeout(function() {

            el.classList.add("fade-out");

            setTimeout(function() {

              el.classList.remove("fade-out");

              avviaRotazioneZona(gridClass, listaSchermi, index + 1);

            }, 400);

          }, durataItemMs);

          zoneTimers.push(t);

        }

      }


      function caricaMeteoDettagliato(containerId) {

        var lat = 41.8058, lon = 12.6775;

        var CACHE_KEY = "meteo_frascati_cache";

        var CACHE_TIME_KEY = "meteo_frascati_cache_time";

        var TEN_MINUTES = 10 * 60 * 1000;


        var cachedData = localStorage.getItem(CACHE_KEY);

        var cachedTime = localStorage.getItem(CACHE_TIME_KEY);

        var now = new Date().getTime();


        if (cachedData && cachedTime && ((now - Number(cachedTime)) < TEN_MINUTES)) {

          try {

            renderizzaWidgetMeteo(JSON.parse(cachedData), containerId);

            return;

          } catch(e) {}

        }


        fetch("https://api.open-meteo.com/v1/forecast?latitude=" + lat + "&longitude=" + lon + "&current=temperature_2m,relative_humidity_2m,weather_code,wind_speed_10m,is_day&daily=weather_code,temperature_2m_max,temperature_2m_min,precipitation_probability_max,wind_speed_10m_max&timezone=auto")

          .then(function(res) { return res.json(); })

          .then(function(data) {

            if (data && data.daily) {

              localStorage.setItem(CACHE_KEY, JSON.stringify(data));

              localStorage.setItem(CACHE_TIME_KEY, now.toString());

              renderizzaWidgetMeteo(data, containerId);

          .catch(function(err) {

      }


      function renderizzaWidgetMeteo(data, containerId) {

        var container = document.getElementById(containerId);

        if (!container || !data.daily) return;


        var oraAttuale = new Date();

        var dataStringa = oraAttuale.toLocaleDateString("it-IT", { weekday: "long", day: "numeric", month: "long" });

        var oraStringa = oraAttuale.toLocaleTimeString("it-IT", { hour: "2-digit", minute: "2-digit" });


        var att = data.current || {};

        var tAttuale = Math.round(att.temperature_2m !== undefined ? att.temperature_2m : data.daily.temperature_2m_max[0]);

        var codeAttuale = Number(att.weather_code !== undefined ? att.weather_code : data.daily.weather_code[0]);

        var isDay = att.is_day !== undefined ? att.is_day : 1;

        var descOggi = getDescrizioneMeteoWMO(codeAttuale, isDay);

        var tMin = data.daily.temperature_2m_min;

        var giorniNomi = ["Dom", "Lun", "Mar", "Mer", "Gio", "Ven", "Sab"];

            '</div>' +

            '<span class="badge bg-light text-dark border px-3 py-2 fs-6"><i class="ti ti-clock me-1 text-muted"></i>' + oraStringa + '</span>' +

      function getSimboloMeteoWMO(code) {

        code = Number(code);

        if (code === 0) return { simbolo: "☀", colore: "#eab308" };

        if (code === 1 || code === 2) return { simbolo: "🌤", colore: "#64748b" };

        if (code === 3) return { simbolo: "☁", colore: "#64748b" };

        if (code === 45 || code === 48) return { simbolo: "≋", colore: "#64748b" };

        if (code >= 51 && code <= 67) return { simbolo: "🌧", colore: "#2563eb" };

        if (code >= 71 && code <= 77) return { simbolo: "❄", colore: "#0ea5e9" };

        if (code >= 80 && code <= 82) return { simbolo: "🌦", colore: "#2563eb" };

        if (code >= 95 && code <= 99) return { simbolo: "⚡", colore: "#7c3aed" };

        return { simbolo: "☁", colore: "#64748b" };

      }


      function getDescrizioneMeteoWMO(code, isDay) {

        if (code === 0 || code === 1) return isDay === 0 ? "Notte Serena" : "Sereno";

        if (code === 2) return isDay === 0 ? "Notte Poco Nuvolosa" : "Poco Nuvoloso";

        if (code === 3) return "Coperto";

        if (code >= 45 && code <= 48) return "Nebbia";

        if (code >= 51 && code <= 67) return "Pioggia";

        if (code >= 71 && code <= 77) return "Neve";

        if (code >= 80 && code <= 82) return "Rovesci";

        if (code >= 95) return "Temporale";

        return "Variabile";

      }

      function escapeHtmlCorsi(str) {

        if (!str) return '';

        return String(str).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;").replace(/"/g, "&quot;").replace(/'/g, "&#039;");

      }


      function pianificaProssimoStep(durataMs) {

        if (stepTimer) clearTimeout(stepTimer);

        stepTimer = setTimeout(function() {

          playlistIndex++;

          if (playlistIndex >= playlistTotale.length) {

            playlistIndex = 0;

            aggiornaPlaylistSilenziosamente(function() {

            });

          } else {

            avviaProssimoStepRotazione();

          }

        }, durataMs);

      }
